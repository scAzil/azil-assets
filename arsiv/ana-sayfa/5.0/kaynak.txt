  }
  function tick(now){
    frameId=0;
    if(!isPageVisible||reduce||!starsOn)return;
    var fps=window.innerWidth<640?24:30;
    if(now-lastFrame>=1000/fps){
      lastFrame=now;
      for(var i=0;i<scenes.length;i++)if(scenes[i].visible)draw(scenes[i],now);
    }
    schedule();
  }
  function schedule(){
    if(frameId||reduce||!isPageVisible||!starsOn)return;
    for(var i=0;i<scenes.length;i++)if(scenes[i].visible){frameId=requestAnimationFrame(tick);break;}
  }
  for(var i=0;i<canvases.length;i++){
    var c=canvases[i],el=c.parentNode,ctx=c.getContext&&c.getContext('2d',{alpha:true});
    if(!ctx)continue;
    var s={el:el,canvas:c,ctx:ctx,type:c.getAttribute('data-network'),seed:i*73+11,w:0,h:0,visible:false,mouse:{x:0,y:0,on:false},nodes:[]};
    scenes.push(s);sizeScene(s);
    (function(scene){
      scene.el.addEventListener('pointermove',function(ev){
        if(ev.pointerType==='touch'||reduce)return;
        var box=scene.el.getBoundingClientRect();
        scene.mouse.x=ev.clientX-box.left;
        scene.mouse.y=ev.clientY-box.top;
        scene.mouse.on=true;
      },{passive:true});
      scene.el.addEventListener('pointerleave',function(){scene.mouse.on=false;},{passive:true});
      if(window.ResizeObserver){new ResizeObserver(function(){sizeScene(scene);}).observe(scene.el);}
      else window.addEventListener('resize',function(){sizeScene(scene);},{passive:true});
    })(s);
  }
  if(window.IntersectionObserver){
    var observer=new IntersectionObserver(function(entries){
      entries.forEach(function(e){
        for(var j=0;j<scenes.length;j++)if(scenes[j].el===e.target)scenes[j].visible=e.isIntersecting;
      });schedule();
      if(!scenes.some(function(scene){return scene.visible;})&&frameId){cancelAnimationFrame(frameId);frameId=0;}
    },{threshold:0});
    for(var k=0;k<scenes.length;k++)observer.observe(scenes[k].el);
  }else{
    for(var n=0;n<scenes.length;n++)scenes[n].visible=true;
    schedule();
  }
  function updateMotion(){
    reduce=!!((motion&&motion.matches)||(connection&&connection.saveData));
    if(frameId){cancelAnimationFrame(frameId);frameId=0;}
    for(var i=0;i<scenes.length;i++){scenes[i].mouse.on=false;draw(scenes[i],0);}
    schedule();
  }
  if(motion){if(motion.addEventListener)motion.addEventListener('change',updateMotion);else if(motion.addListener)motion.addListener(updateMotion);}
  if(connection&&connection.addEventListener)connection.addEventListener('change',updateMotion);
  document.addEventListener('visibilitychange',function(){
    isPageVisible=!document.hidden;
    if(!isPageVisible&&frameId){cancelAnimationFrame(frameId);frameId=0;}
    if(isPageVisible)schedule();
  },{passive:true});
  root.addEventListener('azil:stars',function(){
    starsOn=root.getAttribute('data-stars')!=='off';
    if(!starsOn){
      if(frameId){cancelAnimationFrame(frameId);frameId=0;}
      for(var q=0;q<scenes.length;q++){var item=scenes[q];item.ctx.clearRect(0,0,item.w,item.h);}
    }else{
      for(var k2=0;k2<scenes.length;k2++)draw(scenes[k2],0);
      schedule();
    }
  });
})();
</script>
